bool solve(int g[9][9]) {
  for (int r = 0; r < 9; r++)
    for (int c = 0; c < 9; c++) {
      if (g[r][c]) continue;
      for (int v = 1; v <= 9; v++)
        if (isSafe(g, r, c, v)) {
          g[r][c] = v;
          if (solve(g)) return true;
          g[r][c] = 0; // backtrack
        }
      return false;
    }
  return true;
}
